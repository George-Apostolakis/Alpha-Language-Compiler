cmake_minimum_required(VERSION 3.15)
project(phase2 C)

set(CMAKE_C_STANDARD 99)

find_package(FLEX REQUIRED)
find_package(BISON REQUIRED)

BISON_TARGET(Parser
        ${CMAKE_CURRENT_SOURCE_DIR}/parser.y
        ${CMAKE_CURRENT_BINARY_DIR}/parser.tab.c
        DEFINES_FILE ${CMAKE_CURRENT_BINARY_DIR}/parser.tab.h
)

FLEX_TARGET(Lexer
        ${CMAKE_CURRENT_SOURCE_DIR}/alpha_lexer.l
        ${CMAKE_CURRENT_BINARY_DIR}/alpha_lexer.c
)

ADD_FLEX_BISON_DEPENDENCY(Lexer Parser)

include_directories(
        ${CMAKE_CURRENT_SOURCE_DIR}
        ${CMAKE_CURRENT_BINARY_DIR}
)

add_executable(al2
        ${CMAKE_CURRENT_SOURCE_DIR}/al.c
        ${CMAKE_CURRENT_SOURCE_DIR}/Phase1Functions.c
        ${CMAKE_CURRENT_SOURCE_DIR}/Phase2Functions.c
        ${FLEX_Lexer_OUTPUTS}
        ${BISON_Parser_OUTPUTS}
)

target_compile_options(al2 PRIVATE -Wall -Wextra -g)

target_compile_options(al2 PRIVATE -Wall -Wextra -g -fsanitize=address)
target_link_options(al2 PRIVATE -fsanitize=address)
